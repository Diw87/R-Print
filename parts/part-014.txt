  return new Intl.NumberFormat('pt-BR',{style:'currency',currency:'BRL'}).format(Number(v)||0);

function salvarProdutoMedida(){
  const nome=document.getElementById('medProdutoNome').value.trim();
  const largura=Number(document.getElementById('medLargura').value)||0;
  const altura=Number(document.getElementById('medAltura').value)||0;
  const custoUnit=Number(document.getElementById('medCustoUnit').value)||0;
  const precoUnit=Number(document.getElementById('medPrecoUnit').value)||0;
  const tipo=tipoMedidaAtual();
  const area=largura*altura;

  if(!nome){alert('Digite o nome do produto.');return}
  if(largura<=0 || altura<=0){alert('Informe largura e altura válidas.');return}

  const lista=lerProdutosMedida();
  const agora=new Date().toISOString();

  const item={
    id:produtoMedidaEditandoId||Date.now(),
    nome,tipo,largura,altura,area,
    custoUnit,precoUnit,
    custoTotal:area*custoUnit,
    vendaTotal:area*precoUnit,
    atualizadoEm:agora
  };

  const i=lista.findIndex(p=>Number(p.id)===Number(item.id));
  if(i>=0) lista[i]=item; else lista.push(item);
  gravarProdutosMedida(lista);

  fecharModalProdutoMedida();
  renderizarProdutosAdmin();
  alert('Produto salvo com sucesso!');
}

function normalizarTelefoneWhatsApp(contato){
  let n=String(contato||'').replace(/\D/g,'');
  if(!n) return '';
  // Assume Brasil when DDI is not present.
  if(n.length===10 || n.length===11){
    n='55'+n;
  }
  return n;
}

async function enviarOrcamentoWhatsApp(id){
  const x=await dbGet(id);
  if(!x){
    alert('Orçamento não encontrado.');
    return;
  }

  const telefone=normalizarTelefoneWhatsApp(x.contato);
  if(!telefone){
    alert('Este orçamento não possui um contato válido para WhatsApp.');
    return;
  }

  const numero=x.numero||'';
  const cliente=x.cliente||'';
  const total=moeda(Number(x.total)||0);

async function fazerBackup(){
  const dados=montarDadosBackup();
  const json=JSON.stringify(dados,null,2);
  const nome='RPRINT_Backup_'+new Date().toLocaleDateString('pt-BR').replaceAll('/','-')+'.json';

  // Chrome/Edge modernos: permite escolher local e nome do arquivo.
  if(window.showSaveFilePicker){
    try{
      const handle=await window.showSaveFilePicker({
        suggestedName:nome,
        types:[{
          description:'Backup R-PRINT',
          accept:{'application/json':['.json']}
        }]
      });
      const gravavel=await handle.createWritable();
      await gravavel.write(json);
      await gravavel.close();
      alert('Backup salvo com sucesso!');
      return;
    }catch(e){
      if(e && e.name==='AbortError') return;
      console.warn('Seletor de arquivo indisponível, usando download normal.',e);
    }
  }

  // Fallback: download normal.
  const blob=new Blob([json],{type:'application/json'});
  const url=URL.createObjectURL(blob);
  const a=document.createElement('a');
  a.href=url;
  a.download=nome;
  document.body.appendChild(a);
  a.click();
  a.remove();
  setTimeout(()=>URL.revokeObjectURL(url),1000);
  alert('Backup gerado. Se necessário, mova o arquivo para a sua pasta de backup.');
}

async function processarArquivoBackup(event){
  const arquivo=event.target.files?.[0];
  if(!arquivo) return;

  try{
    const texto=await arquivo.text();
    const dados=JSON.parse(texto);

    if(!dados || !Array.isArray(dados.clientes) || !Array.isArray(dados.produtos) || !Array.isArray(dados.orcamentos)){
      alert('Este arquivo não parece ser um backup válido da R-PRINT.');
      return;
    }

    const confirmar=confirm(
      'Restaurar este backup irá substituir as listas atuais de clientes, produtos e orçamentos. Deseja continuar?'
    );
    if(!confirmar) return;

    gravarClientesLocal(dados.clientes);
    gravarProdutosMedida(dados.produtos);
    gravarOrcamentosLocal(dados.orcamentos);

    await carregarClientesCache();
    await renderizarClientes();
    renderizarProdutosAdmin();
    await renderizarHistorico();

    alert('Backup restaurado com sucesso!');
  }catch(e){
    console.error(e);
    alert('Não foi possível restaurar este arquivo de backup.');
  }
}

function formatarCelularBR(valor){
  let n=String(valor||'').replace(/\D/g,'').slice(0,11);

  if(n.length===0) return '';
  if(n.length<=2) return '('+n;
  if(n.length<=7) return '('+n.slice(0,2)+') '+n.slice(2);

  if(n.length<=10){
    return '('+n.slice(0,2)+') '+n.slice(2,6)+'-'+n.slice(6);
  }

  return '('+n.slice(0,2)+') '+n.slice(2,7)+'-'+n.slice(7);
}
