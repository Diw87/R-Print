function adicionarLaserAoOrcamento(){
  const r=calcularLaser();
  const desc=(document.getElementById('laserDescricao').value||'').trim() || `${document.getElementById('laserOperacao').value} a Laser`;
  if(typeof produtos==='undefined'){alert('Não foi possível acessar o orçamento atual.');return}
  produtos.push({
    id:Date.now(),
    descricao:desc,
    qtd:r.qtd,
    custo:r.custoTotal/r.qtd,
    venda:r.precoSugerido/r.qtd,
    custoTotal:r.custoTotal,
    total:r.precoSugerido,
    origemLaser:true,
    materialLaser:(()=>{
      const sel=document.getElementById('laserMaterialSelecionado');
      const m=typeof lerMateriais==='function'?lerMateriais().find(x=>String(x.id)===String(sel?.value)):null;
      return m?m.nome:'';
    })(),
    materialAreaM2:Number(laserSvgMaterialMedidas.areaM2)||0,
    fitaDuplaFaceAtiva:!!laserFitaEstado.ativa,
    fitaDuplaFaceMetros:Number(laserFitaEstado.metros)||0,
    fitaDuplaFaceCusto:Number(laserFitaEstado.custo)||0,
    fitaDuplaFacePecas:Number(laserFitaEstado.pecasEstimadas)||0,
    fitaDuplaFacePecaMm:Number(laserFitaEstado.comprimentoPecaMm)||0
  });
  if(typeof renderProdutos==='function')renderProdutos();
  if(typeof mostrarNovoOrcamento==='function')mostrarNovoOrcamento();
}
