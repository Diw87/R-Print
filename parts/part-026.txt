(function(){

  const isiOS=()=>/iphone|ipad|ipod/i.test(navigator.userAgent);
  function atualizarBotao(){
    if(!btn)return;
    if(standalone()){btn.classList.remove('show');return;}
    if(installPrompt || isiOS()) btn.classList.add('show');
  }
  window.addEventListener('beforeinstallprompt',function(e){
    e.preventDefault();installPrompt=e;atualizarBotao();
  });
  window.addEventListener('appinstalled',function(){
    installPrompt=null;if(btn)btn.classList.remove('show');
  });
  if(btn)btn.addEventListener('click',async function(){
    if(installPrompt){
      installPrompt.prompt();
      try{await installPrompt.userChoice}catch(e){}
      installPrompt=null;atualizarBotao();return;
    }
    if(isiOS()){
      alert('No iPhone/iPad: toque em Compartilhar e depois em “Adicionar à Tela de Início”.');
      return;
    }
    alert('Abra o menu do navegador e escolha “Instalar R-Print” ou “Adicionar à tela inicial”.');
  });
  if('serviceWorker' in navigator){
    window.addEventListener('load',function(){
      navigator.serviceWorker.register('./sw.js').catch(function(e){console.warn('Service Worker:',e)});
    });
  }
  setTimeout(atualizarBotao,500);
})();
