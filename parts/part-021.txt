function ultimoDiaMes(ano,mesZero){
  return new Date(ano,mesZero+1,0).getDate();
}
function dataRecorrenteISO(ano,mesZero,dia){
  const d=Math.min(Math.max(1,Number(dia)||1),ultimoDiaMes(ano,mesZero));
  return `${ano}-${String(mesZero+1).padStart(2,'0')}-${String(d).padStart(2,'0')}`;
}
function chaveMesISO(dataISO){
  return String(dataISO||'').slice(0,7);
}

/* Gera somente ocorrências que ainda não existem.
   Mantém cada mês independente para poder marcar como paga separadamente. */
function garantirDespesasRecorrentesParaMes(dataRef){
  if(!(dataRef instanceof Date) || isNaN(dataRef))return;
  const lista=lerDespesas();
  const series=new Map();

  lista.filter(d=>d.recorrente && d.serieRecorrenteId).forEach(d=>{
    if(!series.has(d.serieRecorrenteId))series.set(d.serieRecorrenteId,d);
  });

  if(!series.size)return;

  const ano=dataRef.getFullYear(), mes=dataRef.getMonth();
  const chave=`${ano}-${String(mes+1).padStart(2,'0')}`;
  let mudou=false;

  series.forEach(base=>{
    const existe=lista.some(d=>d.serieRecorrenteId===base.serieRecorrenteId && chaveMesISO(d.vencimento)===chave);
    if(existe)return;

    const dia=Number(base.diaVencimento)||Number(String(base.vencimento||'').slice(8,10))||1;
    lista.push({
      ...base,
      id:Date.now()+Math.floor(Math.random()*100000),
      vencimento:dataRecorrenteISO(ano,mes,dia),
      status:'Pendente',
      dataPagamento:'',
      atualizadoEm:new Date().toISOString(),
      geradaAutomaticamente:true
    });
    mudou=true;
  });

  if(mudou)salvarDespesas(lista);
}

function garantirDespesasRecorrentesProximas(){
  const hoje=new Date();
  garantirDespesasRecorrentesParaMes(new Date(hoje.getFullYear(),hoje.getMonth(),1));
  garantirDespesasRecorrentesParaMes(new Date(hoje.getFullYear(),hoje.getMonth()+1,1));
}

  let diaVencimento=Number(document.getElementById('despesaDiaVencimento').value)||Number(String(vencimento||'').slice(8,10))||1;

function mesAtualDashboard(){
  const d=new Date();

function normalizarMesDashboard(valor){
  if(!valor)return mesAtualDashboard();
  const partes=String(valor).split('-');
  if(partes.length!==2)return mesAtualDashboard();
  return new Date(Number(partes[0]),Number(partes[1])-1,1);
}

}

function diasAteVencimento(dataISO){
  if(!dataISO)return null;
  const hoje=new Date();
  hoje.setHours(0,0,0,0);
  const venc=new Date(dataISO+'T12:00:00');
  venc.setHours(0,0,0,0);
  return Math.round((venc-hoje)/86400000);
}

function despesasComAlerta(){
  garantirDespesasRecorrentesProximas();
  return lerDespesas()
    .filter(d=>d.status!=='Paga' && d.vencimento)
    .map(d=>({...d,dias:diasAteVencimento(d.vencimento)}))
    .filter(d=>d.dias!==null && d.dias>=0 && d.dias<=2)
    .sort((a,b)=>a.dias-b.dias || String(a.vencimento).localeCompare(String(b.vencimento)));
}

  const alertas=despesasComAlerta();
  const card=document.getElementById('dashAlertaDespesasCard');

  const txt=document.getElementById('dashAlertasDespesasTexto');

    card.classList.toggle('has-alert',alertas.length>0);
  }

  if(txt){
    if(!alertas.length){

    }else{
      const hoje=alertas.filter(a=>a.dias===0).length;

      if(dois)partes.push(`${dois} em 2 dias`);
      txt.textContent=partes.join(' • ');
    }
  }
}

const RPRINT_META_LUCRO_KEY='rprint_meta_lucro_percentual_v1';

function obterMetaLucroDashboard(){
  const salvo=Number(localStorage.getItem(RPRINT_META_LUCRO_KEY));
  return Number.isFinite(salvo) && salvo>=0 ? salvo : 150;
}

function salvarMetaLucroDashboard(valor){
  let meta=Number(valor);
  if(!Number.isFinite(meta) || meta<0)meta=150;
  localStorage.setItem(RPRINT_META_LUCRO_KEY,String(meta));
  const input=document.getElementById('metaLucroPercentual');
  if(input)input.value=meta;
  atualizarDashboardFinanceiro();
}

function atualizarMetaLucroDashboard(faturamentoMes,despesasMes){
  const metaPct=obterMetaLucroDashboard();
  const input=document.getElementById('metaLucroPercentual');
  if(input)input.value=metaPct;

  const receita=Number(faturamentoMes)||0;
  const despesas=Number(despesasMes)||0;
  const lucroAtual=receita-despesas;

  // Regra adotada:
  // Meta de lucro (%) = lucro desejado como percentual das despesas.
  // Ex.: despesas de R$ 1.000 e meta de 150% => lucro desejado de R$ 1.500
  // e faturamento necessário de R$ 2.500.
  const lucroMeta=despesas*(metaPct/100);
  const faturamentoMeta=despesas+lucroMeta;

  let progresso;
  if(lucroMeta<=0){
    progresso=receita>0 ? 100 : 0;
  }else{
    progresso=(lucroAtual/lucroMeta)*100;
  }
  const progressoVisual=Math.max(0,Math.min(100,progresso));
  const falta=Math.max(0,faturamentoMeta-receita);
  const atingiu=falta<=0 && faturamentoMeta>0;

  const el=(id)=>document.getElementById(id);
  if(el('profitGoalReceita'))el('profitGoalReceita').textContent=moedaFinance(receita);
  if(el('profitGoalDespesas'))el('profitGoalDespesas').textContent=moedaFinance(despesas);
  if(el('profitGoalLucroAtual'))el('profitGoalLucroAtual').textContent=moedaFinance(lucroAtual);
  if(el('profitGoalLucroMeta'))el('profitGoalLucroMeta').textContent=moedaFinance(lucroMeta);
  if(el('profitGoalFaturamentoMeta'))el('profitGoalFaturamentoMeta').textContent=moedaFinance(faturamentoMeta);
  if(el('profitGoalFalta'))el('profitGoalFalta').textContent=atingiu?'Meta atingida':moedaFinance(falta);
  if(el('profitGoalFaltaLabel'))el('profitGoalFaltaLabel').textContent=atingiu?'Situação':'Falta faturar';

  const ring=el('profitGoalRing');
  if(ring){
    const graus=progressoVisual*3.6;
    ring.style.background=`conic-gradient(${atingiu?'#16a34a':'#7c3aed'} ${graus}deg,#eceaf2 ${graus}deg)`;
  }
  if(el('profitGoalPercent')){
    el('profitGoalPercent').textContent=
      `${Math.max(0,progresso).toLocaleString('pt-BR',{maximumFractionDigits:1})}%`;
  }

  const rem=document.querySelector('.profit-goal-row.remaining');
  if(rem)rem.classList.toggle('achieved',atingiu);

  if(el('profitGoalExplanation')){
    if(despesas<=0){
      el('profitGoalExplanation').textContent=
        `Meta configurada em ${metaPct}% de lucro sobre as despesas. Cadastre ou marque despesas como pagas para calcular a meta do mês.`;
    }else{
      el('profitGoalExplanation').textContent=
        `Com ${moedaFinance(despesas)} de despesas e meta de ${metaPct}% de lucro, o lucro desejado é ${moedaFinance(lucroMeta)}. Para isso, o faturamento do mês precisa chegar a ${moedaFinance(faturamentoMeta)}.`;
    }
  }
}

function atualizarDashboardFinanceiro(){
  if(!dashboardDataSelecionada)dashboardDataSelecionada=mesAtualDashboard();

  const sel=new Date(dashboardDataSelecionada.getFullYear(),dashboardDataSelecionada.getMonth(),1);
  const atual=mesAtualDashboard();

  // Proteção para não exibir meses futuros.
  if(sel>atual)dashboardDataSelecionada=atual;

  const ref=dashboardDataSelecionada;
  garantirDespesasRecorrentesParaMes(ref);
  atualizarNomeBotaoMesDashboard(ref);
  const inicioMes=new Date(ref.getFullYear(),ref.getMonth(),1);
  const fimMes=new Date(ref.getFullYear(),ref.getMonth()+1,0,23,59,59);
  const inicioAno=new Date(ref.getFullYear(),0,1);
  const fimAno=new Date(ref.getFullYear(),11,31,23,59,59);

  const fatMes=pagamentosNoPeriodo(inicioMes,fimMes).reduce((s,p)=>s+Number(p.valor||0),0);
