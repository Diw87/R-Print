  document.getElementById('pqData').textContent=
    'Data de Emissão: '+agora.toLocaleDateString('pt-BR')+' - '+

  document.getElementById('pqObs').textContent=observacoes || '';

  const tbody=document.getElementById('pqProdutos');
  if(produtos.length===0){
    tbody.innerHTML='<tr><td colspan="4">Nenhum produto adicionado</td></tr>';
  }else{
    tbody.innerHTML=produtos.map(p=>{
      const precoUnitarioPdf = p.calculoArea ? (Number(p.total)||0) : (Number(p.venda)||0);
      return `
      <tr>
        <td>${escapar(p.descricao)}</td>
        <td>${moeda(precoUnitarioPdf)}</td>
        <td class="num">${p.qtd}</td>
        <td>${moeda(p.total)}</td>
      </tr>`;
    }).join('');
  }

  document.getElementById('pqSubtotal').textContent=moeda(subtotal);
  document.getElementById('pqTotal').textContent=moeda(subtotal);

  // Usa o título da página como nome sugerido do PDF.
  // Exemplo: Prefeitura ODC - N-24
  const tituloAnterior=document.title;

  const nomeClienteArquivo=(cliente || 'Cliente')
    .replace(/[\\/:*?"<>|]+/g,' ')
    .replace(/\s+/g,' ')
    .trim();

  const numeroArquivo=(numero || 'Orcamento')
    .replace(/[\\/:*?"<>|]+/g,' ')
    .replace(/\s+/g,' ')
    .trim();

  document.title=`${nomeClienteArquivo} - ${numeroArquivo}`;

  const restaurarTitulo=()=>{ document.title=tituloAnterior; };
  window.addEventListener('afterprint',restaurarTitulo,{once:true});

  setTimeout(()=>window.print(),100);
}



const ORCAMENTOS_LOCAL_KEY='rprint_orcamentos_v2';
let orcamentoEditandoId=null;

function lerOrcamentosLocal(){
  try{
    const dados=JSON.parse(localStorage.getItem(ORCAMENTOS_LOCAL_KEY)||'[]');
    return Array.isArray(dados)?dados:[];
  }catch(e){
    return [];
  }
}
function gravarOrcamentosLocal(lista){
  localStorage.setItem(ORCAMENTOS_LOCAL_KEY,JSON.stringify(lista));
}
async function dbPut(item){
  const lista=lerOrcamentosLocal();

  if(item.id){
    const id=Number(item.id);
    const i=lista.findIndex(x=>Number(x.id)===id);
    if(i>=0){
      lista[i]={...item,id};
    }else{
      lista.push({...item,id});
    }
    gravarOrcamentosLocal(lista);
    return id;
  }

  const id=Date.now();
  lista.push({...item,id});
  gravarOrcamentosLocal(lista);
  return id;
}
async function dbGetAll(){
  return lerOrcamentosLocal();
}
async function dbGet(id){
  return lerOrcamentosLocal().find(x=>Number(x.id)===Number(id))||null;
}
async function dbDelete(id){
  const lista=lerOrcamentosLocal().filter(x=>Number(x.id)!==Number(id));
  gravarOrcamentosLocal(lista);
}

async function salvarOrcamentoHistorico(){
  const cliente=document.getElementById('cliente').value.trim();
  const contato=document.getElementById('contato').value.trim();
  const numero=document.getElementById('numeroOrcamento').value.trim() || 'N-1';
  const observacoes=(document.getElementById('observacoes')?.value||'').trim();
  const subtotal=produtos.reduce((s,p)=>s+(Number(p.total)||0),0);

  if(!cliente && produtos.length===0){
    alert('Preencha o cliente ou adicione pelo menos um produto antes de salvar.');
    return;
  }

  const agora=new Date().toISOString();
  let criadoEm=agora;

  if(orcamentoEditandoId){
    const antigo=await dbGet(orcamentoEditandoId);
    if(antigo?.criadoEm) criadoEm=antigo.criadoEm;
  }

  const item={
    id:orcamentoEditandoId||undefined,
    numero,
    cliente,
    contato,
    observacoes,
    produtos:JSON.parse(JSON.stringify(produtos)),
    subtotal,
    total:subtotal,
    criadoEm,
    atualizadoEm:agora
  };

  try{
    const id=await dbPut(item);
    orcamentoEditandoId=id;
    await renderizarHistorico();
    alert('Orçamento salvo com sucesso!');
  }catch(e){
    console.error(e);
    alert('Não foi possível salvar o orçamento neste navegador.');
  }
}
async function renderizarHistorico(){
  const list=document.getElementById('savedList');
  if(!list) return;
  let itens=await dbGetAll();

  if(q){
    itens=itens.filter(x=>
      String(x.numero||'').toLowerCase().includes(q) ||

    return;
  }

  list.innerHTML=`
    <div class="saved-row header">
      <div>Número</div><div>Cliente</div><div>Atualizado</div><div>Total</div><div style="text-align:right">Ações</div>
    </div>`+

          <button class="btn btn-outline" onclick="duplicarOrcamento(${x.id})">Duplicar</button>

async function mostrarSugestoesClientes(){
  if(!clientesCache.length) await carregarClientesCache();

  const input=document.getElementById('cliente');
  const box=document.getElementById('clientSuggestions');
  const q=(input.value||'').trim().toLowerCase();

  if(!q){
    box.classList.remove('show');
    box.innerHTML='';
    return;
  }

  const achados=clientesCache
    .filter(c=>String(c.nome||'').toLowerCase().startsWith(q))
    .slice(0,8);

  if(!achados.length){
    box.classList.remove('show');
    box.innerHTML='';
    return;
  }

  box.innerHTML=achados.map(c=>`
    <button type="button" class="client-suggestion" onclick="selecionarCliente(${c.id})">
      <strong>${escapar(c.nome||'')}</strong>
      <small>${escapar(c.contato||'Sem contato cadastrado')}</small>
    </button>
  `).join('');
  box.classList.add('show');
}

async function selecionarCliente(id){

  document.getElementById('cliente').value=c.nome||'';
  document.getElementById('contato').value=c.contato||'';
  const box=document.getElementById('clientSuggestions');
  box.classList.remove('show');
  box.innerHTML='';
}
