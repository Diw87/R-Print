async function salvarNovoUsuarioLocal(event){
  event.preventDefault();
  const nome=document.getElementById('newLocalUserName').value.trim();
  const senha=document.getElementById('newLocalUserPass').value;
  const senha2=document.getElementById('newLocalUserPass2').value;
  const dica=(document.getElementById('newLocalUserHint').value||'').trim();
  const msg=document.getElementById('newLocalUserMsg');
  msg.textContent='';

  if(!nome){
    msg.textContent='Digite o nome do usuário.';
    return;
  }
  if(senha.length<4){
    msg.textContent='A senha precisa ter pelo menos 4 caracteres.';
    return;
  }
  if(senha!==senha2){
    msg.textContent='As senhas não conferem.';
    return;
  }

  const users=rprintGetUsers();
  if(users.some(u=>u.user.toLowerCase()===nome.toLowerCase())){
    msg.textContent='Esse nome de usuário já existe.';
    return;
  }

  users.push({
    id:Date.now(),
    user:nome,
    passHash:await rprintHash(senha),
    hint:dica,
    criadoEm:new Date().toISOString()
  });

  rprintSaveUsers(users);
  document.getElementById('newLocalUserName').value='';
  document.getElementById('newLocalUserPass').value='';
  document.getElementById('newLocalUserPass2').value='';
  document.getElementById('newLocalUserHint').value='';
  renderUsuariosLocais();
  alert('Novo usuário adicionado com sucesso.');
}

function renderUsuariosLocais(){
  const box=document.getElementById('localUsersList');
  if(!box)return;

  const users=rprintGetUsers();
  const atual=sessionStorage.getItem(RPRINT_LOCAL_SESSION_KEY);

  if(!users.length){
    box.innerHTML='<div class="empty-history">Nenhum usuário cadastrado.</div>';
    return;
  }

  box.innerHTML=users.map(u=>`
    <div class="local-user-row">
      <div class="local-user-row-profile">
        <span class="local-user-list-avatar">
          ${u.photo?`<img src="${u.photo}" alt="">`:`<svg viewBox="0 0 24 24"><circle cx="12" cy="8" r="3.5"/><path d="M5 20c.7-4 3.1-6 7-6s6.3 2 7 6"/></svg>`}
        </span>
        <div>
          <strong>${escapar(u.user||'Usuário')}
            ${u.user===atual?'<span class="local-user-current">Você</span>':''}
          </strong>
          <small>${u.hint?'Dica cadastrada':'Sem dica de senha'}</small>
        </div>
      </div>
      <div class="local-user-row-actions">
        ${u.user!==atual?`<button type="button" class="local-user-delete" onclick="excluirUsuarioLocal(${u.id})">Excluir</button>`:''}
      </div>
    </div>

function excluirUsuarioLocal(id){
  const users=rprintGetUsers();
  const atual=sessionStorage.getItem(RPRINT_LOCAL_SESSION_KEY);
  const alvo=users.find(u=>Number(u.id)===Number(id));
  if(!alvo)return;

  if(alvo.user===atual){
    alert('Você não pode excluir o usuário que está conectado.');
    return;
  }

  if(users.length<=1){
    alert('É necessário manter pelo menos um usuário cadastrado.');
    return;
  }

  if(!confirm(`Deseja excluir o usuário "${alvo.user}"?`))return;

  rprintSaveUsers(users.filter(u=>Number(u.id)!==Number(id)));
  renderUsuariosLocais();
}

async function aprovarOrcamentoComoPedido(id){
  const orc=await dbGet(id);
  if(!orc){
    alert('Orçamento não encontrado.');
    return;
  }

  const pedidos=lerPedidosLocal();
  if(pedidos.some(p=>Number(p.orcamentoId)===Number(id))){

    return;
  }

  const pedido={
    id:Date.now(),
    orcamentoId:Number(id),
    numero:orc.numero||'-',
    cliente:orc.cliente||'Cliente não informado',
    contato:orc.contato||'',

function atualizarDataEntregaPedido(id,dataEntrega){
  const pedidos=lerPedidosLocal();

  atualizarAlertaEntregasDashboard();

function diasAteEntregaPedido(dataISO){
  if(!dataISO)return null;
  const hoje=new Date();
  hoje.setHours(0,0,0,0);
  const entrega=new Date(dataISO+'T12:00:00');
  entrega.setHours(0,0,0,0);
  return Math.round((entrega-hoje)/86400000);
}

function pedidosComAlertaEntrega(){
  return lerPedidosLocal()
    .filter(p=>p.status!=='Entregue' && p.dataEntrega)
    .map(p=>({...p,diasEntrega:diasAteEntregaPedido(p.dataEntrega)}))
    .filter(p=>p.diasEntrega!==null && p.diasEntrega<=2)
    .sort((a,b)=>a.diasEntrega-b.diasEntrega || String(a.dataEntrega).localeCompare(String(b.dataEntrega)));
}

function atualizarAlertaEntregasDashboard(){
  const alertas=pedidosComAlertaEntrega();
  const card=document.getElementById('dashAlertaEntregasCard');
  const qtd=document.getElementById('dashAlertasEntregas');
  const txt=document.getElementById('dashAlertasEntregasTexto');

  if(qtd)qtd.textContent=String(alertas.length);
  if(card)card.classList.toggle('has-alert',alertas.length>0);

  if(txt){
    if(!alertas.length){
      txt.textContent='Nenhuma entrega próxima';
    }else{
      const atrasados=alertas.filter(a=>a.diasEntrega<0).length;
      const hoje=alertas.filter(a=>a.diasEntrega===0).length;
      const amanha=alertas.filter(a=>a.diasEntrega===1).length;
      const dois=alertas.filter(a=>a.diasEntrega===2).length;
      const partes=[];
      if(atrasados)partes.push(`${atrasados} atrasado${atrasados>1?'s':''}`);
      if(hoje)partes.push(`${hoje} hoje`);
      if(amanha)partes.push(`${amanha} amanhã`);
      if(dois)partes.push(`${dois} em 2 dias`);
      txt.textContent=partes.join(' • ');
    }
  }
}

  const box=document.getElementById('alertasEntregasLista');

  if(box){
    if(!alertas.length){
      box.innerHTML='<div class="delivery-alert-empty">Nenhum pedido com entrega próxima ou atrasada.</div>';
    }else{
      box.innerHTML=alertas.map(p=>{
        const d=p.diasEntrega;
        const classe=d<0?'overdue':d===0?'today':d===1?'tomorrow':'';
        const textoDias=d<0
          ? `Atrasado ${Math.abs(d)} dia${Math.abs(d)>1?'s':''}`

          : d===1?'Entrega amanhã'
          :'Entrega em 2 dias';

        const entrega=new Date(p.dataEntrega+'T12:00:00').toLocaleDateString('pt-BR');
        return `<div class="delivery-alert-item ${classe}">
          <div>
            <strong>Pedido ${escapar(p.numero||'-')} • ${escapar(p.cliente||'Cliente')}</strong>
            <small>${moeda(p.total||0)} • Entrega ${entrega} • Status: ${escapar(p.status||'Aguardando')}</small>
          </div>

        </div>`;
      }).join('');

function classeStatusPedidoCor(status){
  const s=String(status||'').trim().toLowerCase()
    .normalize('NFD').replace(/[\u0300-\u036f]/g,'');
  if(s.includes('entreg'))return 'pedido-status-entregue';
  if(s.includes('pronto'))return 'pedido-status-pronto';
  if(s.includes('produc'))return 'pedido-status-producao';
  return 'pedido-status-aguardando';
}


const RPRINT_PAGINACAO={};

function estadoPaginacaoLista(chave){
  if(!RPRINT_PAGINACAO[chave])RPRINT_PAGINACAO[chave]={pagina:1,porPagina:20};
  return RPRINT_PAGINACAO[chave];
}
function paginarLista(itens,chave){
  const st=estadoPaginacaoLista(chave);
  const total=itens.length;
  const paginas=Math.max(1,Math.ceil(total/st.porPagina));
  st.pagina=Math.min(Math.max(1,st.pagina),paginas);
  const inicio=(st.pagina-1)*st.porPagina;
  return {
    itens:itens.slice(inicio,inicio+st.porPagina),
    total,
    paginas,
    pagina:st.pagina,
    porPagina:st.porPagina,
    inicio: total ? inicio+1 : 0,
    fim: Math.min(inicio+st.porPagina,total)
  };
}
function mudarPaginaLista(chave,pagina,renderFn){
  const st=estadoPaginacaoLista(chave);
  st.pagina=Math.max(1,Number(pagina)||1);
  const fn=window[renderFn];
  if(typeof fn==='function')fn();
}
function mudarPorPaginaLista(chave,valor,renderFn){
  const st=estadoPaginacaoLista(chave);
  st.porPagina=Math.max(5,Number(valor)||20);
  st.pagina=1;
  const fn=window[renderFn];
  if(typeof fn==='function')fn();
}
function htmlPaginacaoLista(chave,p,renderFn){
  if(!p || p.total<=0)return '';
  return `<div class="rprint-pagination">
    <div class="rprint-pagination-left">
      <span class="rprint-pagination-info">Mostrando <strong>${p.inicio}-${p.fim}</strong> de <strong>${p.total}</strong></span>
      <span>Itens por página:</span>
      <select onchange="mudarPorPaginaLista('${chave}',this.value,'${renderFn}')">
        ${[10,20,50,100].map(n=>`<option value="${n}" ${p.porPagina===n?'selected':''}>${n}</option>`).join('')}
      </select>
    </div>
    <div class="rprint-pagination-right">
      <button class="rprint-page-btn" ${p.pagina<=1?'disabled':''} onclick="mudarPaginaLista('${chave}',${p.pagina-1},'${renderFn}')">‹ Anterior</button>
      <span class="rprint-page-number">${p.pagina}</span>
      <span>de ${p.paginas}</span>
      <button class="rprint-page-btn" ${p.pagina>=p.paginas?'disabled':''} onclick="mudarPaginaLista('${chave}',${p.pagina+1},'${renderFn}')">Próxima ›</button>
    </div>
  </div>`;
}
