function renderCalendarioHome(){
  const meses=['Janeiro','Fevereiro','Março','Abril','Maio','Junho','Julho','Agosto','Setembro','Outubro','Novembro','Dezembro'];
  const y=homeCalendarDate.getFullYear(), m=homeCalendarDate.getMonth();
  document.getElementById('homeCalendarTitle').textContent=meses[m]+' de '+y;
  const first=new Date(y,m,1), start=first.getDay(), days=new Date(y,m+1,0).getDate(), prevDays=new Date(y,m,0).getDate();
  const today=new Date();
  let cells=[];
  for(let i=start-1;i>=0;i--) cells.push(`<div class="calendar-day muted">${prevDays-i}</div>`);
  for(let d=1;d<=days;d++){
    const isToday=d===today.getDate()&&m===today.getMonth()&&y===today.getFullYear();
    cells.push(`<div class="calendar-day ${isToday?'today':''}">${String(d).padStart(2,'0')}</div>`);
  }
  let n=1; while(cells.length<42) cells.push(`<div class="calendar-day muted">${String(n++).padStart(2,'0')}</div>`);
  document.getElementById('homeCalendarDays').innerHTML=cells.join('');
}

async function obterOrcamentosTarefas(){
  try{

  return o.itens || o.items || o.produtos || o.servicos || [];

  return o.numero || o.numero_orcamento || o.id || '-';
}
function clienteOrcamentoTarefa(o){
  return o.cliente || o.cliente_nome || o.nomeCliente || o.nome_cliente || '-';
}
async function renderOrcamentosParaTarefas(){
  const list=document.getElementById('taskQuoteList'); if(!list)return;
  const dados=await obterOrcamentosTarefas();
  const q=(document.getElementById('taskQuoteSearch')?.value||'').trim().toLowerCase();
  const filtrados=dados.filter(o=>{
    const txt=(String(numeroOrcamentoTarefa(o))+' '+String(clienteOrcamentoTarefa(o))).toLowerCase();
    return !q || txt.includes(q);
  });
  if(!filtrados.length){
    list.innerHTML='<div class="task-picker-empty">Nenhum orçamento salvo encontrado.</div>';
    return;
  }
  list.innerHTML=filtrados.slice().reverse().map((o,idx)=>{
    const realIndex=dados.indexOf(o);
    const itens=itensDoOrcamentoTarefa(o);
    return `<div class="task-quote-card">
      <button type="button" class="task-quote-summary" onclick="toggleServicosOrcamentoTarefa(${realIndex},this)">
        <div><strong>Orçamento ${escapar(String(numeroOrcamentoTarefa(o)))}</strong><small>${escapar(String(clienteOrcamentoTarefa(o)))}</small></div>
        <div class="task-quote-count">${itens.length} ${itens.length===1?'item':'itens'} <span>⌄</span></div>
      </button>
      <div class="task-service-list"></div>
    </div>`;
  }).join('');
  window.__rprintTaskQuotes=dados;
}
function toggleServicosOrcamentoTarefa(index,btn){
  const o=(window.__rprintTaskQuotes||[])[index]; if(!o)return;
  const area=btn.parentElement.querySelector('.task-service-list');
  const aberto=area.classList.contains('show');
  document.querySelectorAll('.task-service-list.show').forEach(x=>{x.classList.remove('show');x.innerHTML=''});
  if(aberto)return;
  const itens=itensDoOrcamentoTarefa(o);
  if(!itens.length){
    area.innerHTML='<div class="task-no-services">Esse orçamento não possui itens.</div>';
  }else{
    area.innerHTML=itens.map((item,i)=>{
      const nome=item.produto || item.nome || item.descricao || item.servico || 'Item do orçamento';
      const qtd=item.qtd || item.quantidade || '';
      return `<div class="task-service-option">
        <div>
          <strong>${escapar(String(nome))}</strong>
          ${qtd?`<small>Quantidade: ${escapar(String(qtd))}</small>`:''}
        </div>
        <button type="button" onclick="adicionarServicoDoOrcamento(${index},${i},this)">＋ Adicionar</button>
      </div>`;
    }).join('');
  }
  area.classList.add('show');
}
function adicionarServicoDoOrcamento(orcIndex,itemIndex,btn){
  const o=(window.__rprintTaskQuotes||[])[orcIndex]; if(!o)return;
  const item=itensDoOrcamentoTarefa(o)[itemIndex]; if(!item)return;
  const nome=item.produto || item.nome || item.descricao || item.servico || 'Item do orçamento';
  const qtd=item.qtd || item.quantidade || '';
  const t=getTarefasHome();
  t.push({
    servico:String(nome),
    qtd:qtd,
    numero:String(numeroOrcamentoTarefa(o)),
    cliente:String(clienteOrcamentoTarefa(o)),
    done:false
  });
  salvarTarefasHome(t);
  renderTarefasHome();
  if(btn){
    const old=btn.textContent;
    btn.textContent='✓ Adicionado';
    btn.disabled=true;
    setTimeout(()=>{btn.textContent=old;btn.disabled=false},900);
  }
}
