function atualizarCalculoMaterialModal(){
  const unidade=document.getElementById('materialUnidade')?.value||'m2';
  const valorPeca=Number(document.getElementById('materialValorPeca')?.value)||0;
  const larguraDigitada=Number(document.getElementById('materialLargura')?.value)||0;
  const altura=Number(document.getElementById('materialAltura')?.value)||0;
  const qtdMetros=Number(document.getElementById('materialQtdMetros')?.value)||0;

  const labelLargura=document.getElementById('materialLarguraLabel');
  const labelQtd=document.getElementById('materialQtdMetrosLabel');
  const alturaField=document.getElementById('materialAlturaField');
  const ajuda=document.getElementById('materialRoloAjuda');
  const larguraInput=document.getElementById('materialLargura');

  if(unidade==='m'){
    if(labelLargura)labelLargura.textContent='Largura da fita / rolo (mm)';
    if(labelQtd)labelQtd.textContent='Comprimento do rolo (m)';
    if(alturaField)alturaField.style.display='none';
    if(ajuda)ajuda.classList.add('show');
    if(larguraInput)larguraInput.placeholder='Ex.: 19';
  }else{
    if(labelLargura)labelLargura.textContent='Largura do material (m)';
    if(labelQtd)labelQtd.textContent='Quantidade de metros';
    if(alturaField)alturaField.style.display='';
    if(ajuda)ajuda.classList.remove('show');
    if(larguraInput)larguraInput.placeholder='Ex.: 1,22';
  }

  // Para metro linear, o usuário informa largura em mm.
  // Internamente convertemos para metros para os cálculos.
  const larguraM=unidade==='m' ? larguraDigitada/1000 : larguraDigitada;

  const areaPeca=Math.max(0,larguraM*altura);
  let areaTotal=areaPeca;
  if(larguraM>0 && qtdMetros>0)areaTotal=larguraM*qtdMetros;

  let custoCalculado=0;
  let custoM2=0;

  if(unidade==='m2'){
    const divisor=areaTotal>0?areaTotal:areaPeca;
    custoCalculado=divisor>0 ? valorPeca/divisor : 0;
    custoM2=custoCalculado;
  }else if(unidade==='m'){
    // Valor do rolo ÷ comprimento do rolo = custo por metro linear.
    custoCalculado=qtdMetros>0 ? valorPeca/qtdMetros : 0;
    custoM2=(larguraM>0 && custoCalculado>0) ? custoCalculado/larguraM : 0;
  }else if(unidade==='un'){
    custoCalculado=valorPeca;
    custoM2=areaPeca>0 ? valorPeca/areaPeca : 0;
  }else{
    custoCalculado=valorPeca;
  }

  const custoInput=document.getElementById('materialCusto');
  if(custoInput)custoInput.value=custoCalculado.toFixed(4);

  const areaPecaEl=document.getElementById('materialAreaPeca');
  const areaTotalEl=document.getElementById('materialAreaTotal');
  const custoM2El=document.getElementById('materialCustoM2');

  if(areaPecaEl){
    areaPecaEl.textContent=unidade==='m'
      ? `${larguraDigitada.toLocaleString('pt-BR',{maximumFractionDigits:2})} mm de largura`
      : areaPeca.toLocaleString('pt-BR',{minimumFractionDigits:4,maximumFractionDigits:4})+' m²';
  }

  if(areaTotalEl){
    areaTotalEl.textContent=unidade==='m'
      ? `${qtdMetros.toLocaleString('pt-BR',{maximumFractionDigits:2})} m de rolo`
      : areaTotal.toLocaleString('pt-BR',{minimumFractionDigits:4,maximumFractionDigits:4})+' m²';
  }

  if(custoM2El){
    custoM2El.textContent=unidade==='m'
      ? `${moedaMaterial(custoCalculado)} / m`
      : moedaMaterial(custoM2);
  }
}

function unidadeSvgParaMetros(valor,unidade){
  const v=Number(valor)||0;
  const u=String(unidade||'').toLowerCase();
  if(u==='mm')return v/1000;
  if(u==='cm')return v/100;
  if(u==='in')return v*0.0254;
  if(u==='pt')return v*0.0254/72;
  if(u==='pc')return v*0.0254/6;
  if(u==='m')return v;
  return null;
}
function parseSvgLength(txt){
  const m=String(txt||'').trim().match(/^([+-]?(?:\d+\.?\d*|\.\d+))\s*(mm|cm|in|pt|pc|m|px)?$/i);
  if(!m)return null;
  return {value:Number(m[1]),unit:(m[2]||'px').toLowerCase()};
}
function corSvg(el){
  const style=(el.getAttribute('style')||'').split(';').reduce((a,x)=>{
    const p=x.split(':'); if(p.length===2)a[p[0].trim()]=p[1].trim(); return a;
  },{});
  let fill=el.getAttribute('fill')||style.fill||'';
  fill=String(fill).trim().toLowerCase();
  if(!fill || fill==='none' || fill==='transparent')return null;
  return fill;
}
function areaPoligono(points){
  const pts=String(points||'').trim().split(/\s+/).map(p=>p.split(',').map(Number)).filter(p=>p.length===2&&p.every(Number.isFinite));
  if(pts.length<3)return 0;
