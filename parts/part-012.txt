<script>
let produtos = [];
let acabamentosAtuais = [];

function moeda(v){
  return new Intl.NumberFormat('pt-BR',{style:'currency',currency:'BRL'}).format(Number(v)||0);
}
function numeroBR(v){
  return (Number(v)||0).toLocaleString('pt-BR',{minimumFractionDigits:2,maximumFractionDigits:2});
}
function abrirModal(){
  document.getElementById('modalBackdrop').classList.add('show');
  setTimeout(()=>document.getElementById('produtoBusca').focus(),50);
}
function fecharModal(){
  document.getElementById('modalBackdrop').classList.remove('show');
  limparModal();
}
function fecharSeFora(e){
  if(e.target.id==='modalBackdrop') fecharModal();
}
function alternarManual(){
  const ativo=document.getElementById('manualToggle').checked;
  document.getElementById('manualArea').classList.toggle('show',ativo);
  document.getElementById('produtoBusca').placeholder=ativo
    ? 'Nome curto do produto ou serviço'
    : 'Pesquise pelo nome do produto...';
}
function calcularModal(){
  const qtd=Number(document.getElementById('quantidade').value)||0;
  const custo=Number(document.getElementById('custoProducao').value)||0;
  const venda=Number(document.getElementById('precoVenda').value)||0;
  const desligado=document.getElementById('disableCalc').checked;

  const custoTotal=desligado ? custo : qtd*custo;
  const vendaTotal=desligado ? venda : qtd*venda;

  document.getElementById('custoTotal').value=numeroBR(custoTotal);
  document.getElementById('precoVendaTotal').value=numeroBR(vendaTotal);
}
function adicionarAcabamento(){
  const input=document.getElementById('acabamentoInput');
  const nome=input.value.trim();
  if(!nome)return;
  acabamentosAtuais.push(nome);
  input.value='';
  renderAcabamentos();
}
function renderAcabamentos(){
  document.getElementById('acabamentosList').innerHTML=
    acabamentosAtuais.map((a,i)=>`<span class="tag">${escapar(a)} <button onclick="removerAcabamento(${i})" style="border:0;background:transparent;color:#7c3aed;cursor:pointer">×</button></span>`).join('');
}
function removerAcabamento(i){
  acabamentosAtuais.splice(i,1);
  renderAcabamentos();
}
function adicionarProduto(){
  const descricao=document.getElementById('produtoBusca').value.trim();
  const qtd=Number(document.getElementById('quantidade').value)||0;
  const custo=Number(document.getElementById('custoProducao').value)||0;
  const venda=Number(document.getElementById('precoVenda').value)||0;
  const desligado=document.getElementById('disableCalc').checked;

  if(!descricao){alert('Informe o nome ou a descrição do produto.');return;}
  if(qtd<=0){alert('Informe uma quantidade válida.');return;}

  produtos.push({
    id:Date.now(),
    descricao,
    qtd,
    custo,
    venda,
    custoTotal:desligado?custo:qtd*custo,
    total:desligado?venda:qtd*venda
  });

  renderProdutos();
  fecharModal();
}
function renderProdutos(){
  const tbody=document.getElementById('productRows');

  if(produtos.length===0){
    tbody.innerHTML=`<tr><td colspan="6"><div class="empty">
      <div class="empty-icon">📦</div>
      <strong>Nenhum produto adicionado</strong><br>
      <span class="small-note">Clique em “DESCREVER PRODUTO” para adicionar itens ao orçamento.</span>
    </div></td></tr>`;
  }else{
    tbody.innerHTML=produtos.map((p,i)=>`
      <tr>
        <td>${i+1}</td>
        <td><strong>${escapar(p.descricao)}</strong></td>
        <td>
          <div class="cart-qty-control no-print">
            <button type="button" onclick="alterarQuantidadeCarrinho(${p.id},-1)" aria-label="Diminuir quantidade">−</button>
            <input type="number" min="1" step="1" value="${p.qtd}" onchange="definirQuantidadeCarrinho(${p.id},this.value)" aria-label="Quantidade">
            <button type="button" onclick="alterarQuantidadeCarrinho(${p.id},1)" aria-label="Aumentar quantidade">+</button>
          </div>
          <span class="cart-qty-print">${p.qtd}</span>
        </td>
        <td>${moeda(p.venda)}</td>
        <td><strong>${moeda(p.total)}</strong></td>
        <td class="no-print"><button class="btn btn-danger" onclick="removerProduto(${p.id})">Excluir</button></td>
      </tr>
    `).join('');
  }

  const subtotal=produtos.reduce((s,p)=>s+p.total,0);
  const total=subtotal;
  document.getElementById('subtotal').textContent=moeda(subtotal);
  document.getElementById('grandTotal').textContent=moeda(total);
  document.getElementById('subtotalLabel').textContent=`Subtotal (${produtos.length} ${produtos.length===1?'item':'itens'})`;
}
function removerProduto(id){
  produtos=produtos.filter(p=>p.id!==id);
  renderProdutos();
}
function recalcularQuantidadeProduto(p,novaQtd){
  const qtd=Math.max(1,Math.floor(Number(novaQtd)||1));
  p.qtd=qtd;

  // Produtos comuns: total acompanha QTD × preço unitário.
  // Produtos por m²/cm² também preservam o valor unitário já calculado.
  p.total=qtd*(Number(p.venda)||0);

  // Mantém custo total coerente quando o item possuir custo.
  if(Object.prototype.hasOwnProperty.call(p,'custo')){
    p.custoTotal=qtd*(Number(p.custo)||0);
  }
}
function definirQuantidadeCarrinho(id,valor){
  const p=produtos.find(x=>Number(x.id)===Number(id));
  if(!p)return;
  recalcularQuantidadeProduto(p,valor);
  renderProdutos();
}
function alterarQuantidadeCarrinho(id,delta){
  const p=produtos.find(x=>Number(x.id)===Number(id));
  if(!p)return;
  recalcularQuantidadeProduto(p,(Number(p.qtd)||1)+Number(delta||0));
  renderProdutos();
}
function limparModal(){
  document.getElementById('produtoBusca').value='';
  document.getElementById('produtoBusca').placeholder='Pesquise pelo nome do produto...';
  document.getElementById('quantidade').value=1;
  document.getElementById('custoProducao').value='0.00';
  document.getElementById('precoVenda').value='0.00';
  document.getElementById('disableCalc').checked=false;
  calcularModal();
}
function escapar(t){
  return String(t).replace(/[&<>"']/g,m=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#039;'}[m]));
}

function carregar(){
  const salvo=localStorage.getItem('orcamentoAtualV2');
  if(!salvo)return;
  try{
    const d=JSON.parse(salvo);
    document.getElementById('cliente').value=d.cliente||'';
    document.getElementById('contato').value=d.contato||'';
    produtos=Array.isArray(d.produtos)?d.produtos:[];
    renderProdutos();
  }catch(e){}
}

function imprimirOrcamento(){
  const cliente=(document.getElementById('cliente').value||'').trim() || 'Cliente não informado';
  const contato=(document.getElementById('contato').value||'').trim() || '-';
  const numero=(document.getElementById('numeroOrcamento').value||'').trim() || 'N-1';
  const observacoes=(document.getElementById('observacoes').value||'').trim();
  const subtotal=produtos.reduce((s,p)=>s+(Number(p.total)||0),0);

  const logo=
    document.querySelector('.brand-logo') ||
    document.querySelector('.sidebar-avatar img');
  if(logo && document.getElementById('pqLogo')) {
    document.getElementById('pqLogo').src=logo.src;
  }

  document.getElementById('pqNumero').textContent='Orçamento '+numero;
  const agora=new Date();
